\section{LilyPond 基本语法速查}

\subsection{行内乐谱}

行内乐谱用 \texttt{\string\lilypond\{ ... \}} 命令嵌入正文，随文本流式排版，适用于短小的音乐片段。


\begin{LTXexample}[language=TeX, basicstyle=\ttfamily\small]
\lilypond{c'4 d' e' f'}
\lilypond{<c' e' g'>2.}
\lilypond{g'8 a' b' c''}
\end{LTXexample}

\subsection{常用 inline 示例}

\subsubsection{单音与音阶}

\lilypond{c'4 d' e' f'}，或快速音阶跑动
\lilypond{c''16 d'' e'' f'' g'' a'' b'' c'''}。

\subsubsection{和弦}

单个和弦 \lilypond{<c' e' g'>2.}，或带琶音记号的和弦
\lilypond{<c' e' g'>\arpeggio}。

\subsubsection{连奏与符杠}

\lilypond{c'8( d' e' f')} 为连奏，\lilypond{c'8[ d' e' f']} 为符杠分组。

\subsubsection{休止符}

\lilypond{r4 r8 r16} 表示四分、八分、十六分休止符。

\subsubsection{强弱记号}

\lilypond{c'4\p d'\mf e'\f f'\ff} 展示不同力度标记。

\subsection{与整段乐谱的区别}

\begin{center}
\begin{tabular}{lll}
 & \textbf{Inline} & \textbf{完整乐谱} \\
命令 & \texttt{\string\lilypond\{...\}} & \texttt{\string\begin\{lilypond\}...\string\end\{lilypond\}} \\
适用场景 & 短片段、示例 & 完整乐谱、多声部、多行 \\
换行 & 不换行 & 独立成段 \\
\end{tabular}
\end{center}

\subsection{注意事项}

\begin{enumerate}
\item \texttt{\string\lilypond} 命令必须在一行内完成，不能跨行。
\item 内容较长时建议使用 \texttt{lilypond} 环境而非 inline。
\item 如果 \texttt{\string\lilypond} 命令频繁使用，建议定义简写宏：
\begin{lstlisting}[language=TeX, basicstyle=\ttfamily\small]
\newcommand{\ly}[1]{\lilypond{#1}}
% 之后可用 \ly{c'4 d' e'} 简化输入
\end{lstlisting}
\end{enumerate}

\subsection{音符与时值}

音符用小写字母 \texttt{c d e f g a b} 表示，时值用数字跟在后面：

\begin{center}
\begin{tabular}{lll}
\texttt{c4} & 四分音符 C \\
\texttt{d8} & 八分音符 D \\
\texttt{e2} & 二分音符 E \\
\texttt{f1} & 全音符 F \\
\texttt{g16} & 十六分音符 G \\
\texttt{a4.} & 附点四分音符 A \\
\end{tabular}
\end{center}

八度标记：\texttt{c'} 为中央 C，\texttt{c''} 高八度，\texttt{c,} 低八度。

\subsection{休止符}

休止符用 \texttt{r} 表示，后跟时值：\texttt{r4}、\texttt{r8}、\texttt{r2.}。

\subsection{升降号与调号}

\begin{center}
\begin{tabular}{lll}
\texttt{cis'} & 升 C \\
\texttt{des'} & 降 D \\
\texttt{\string\key f \string\minor} & 设置 f 小调调号 \\
\texttt{\string\key c \string\major} & 设置 C 大调调号 \\
\end{tabular}
\end{center}

\subsection{连音与分组}

\begin{center}
\begin{tabular}{lll}
\texttt{c8( d e)} & 连奏（slur） \\
\texttt{c8[ d e]} & 符杠分组（beam） \\
\texttt{c4\string~ c4} & 延音线（tie） \\
\end{tabular}
\end{center}

\subsection{和弦}

和弦用尖括号括起：\texttt{<c e g>}，后跟时值如 \texttt{<c e g>2.}。琶音记号用 \texttt{\string\arpeggio}。

\subsection{谱面指令}

\begin{center}
\begin{tabular}{lll}
\texttt{\string\clef treble} & 高音谱号 \\
\texttt{\string\clef bass} & 低音谱号 \\
\texttt{\string\stemUp} & 符干朝上 \\
\texttt{\string\stemDown} & 符干朝下 \\
\texttt{\string\voiceOne} & 第一声部 \\
\texttt{\string\voiceTwo} & 第二声部 \\
\end{tabular}
\end{center}

\subsection{强弱与速度}

\begin{center}
\begin{tabular}{lll}
\texttt{\string\p}、\texttt{\string\mp}、\texttt{\string\mf}、\texttt{\string\f}、\texttt{\string\ff} & 强弱记号 \\
\texttt{\string\< ... \string\!} & 渐强 \\
\texttt{\string\> ... \string\!} & 渐弱 \\
\texttt{\string\tempo 4=80} & 速度设定 \\
\texttt{\string\tempo "Andante"} & 速度文字 \\
\end{tabular}
\end{center}

\subsection{基本结构}

\begin{lstlisting}[language=TeX, basicstyle=\ttfamily\small]
\version "2.24.4"

\score {
  \new PianoStaff <<
    \new Staff = "right" \relative c'' {
      \key f \minor
      \time 6/8
      \clef treble
      c4. d8 e f
    }
    \new Staff = "left" \relative c {
      \key f \minor
      \time 6/8
      \clef bass
      f4. f4.
    }
  >>
  \layout { }
}
\end{lstlisting}

\subsection{常见语法陷阱}

\begin{enumerate}
\item 括号必须成对：\texttt{\{} 与 \texttt{\}}、\texttt{<<} 与 \texttt{>>}、\texttt{(} 与 \texttt{)}。
\item \texttt{\string\score} 块必须完整闭合。
\item 命令拼写错误会导致编译失败，如 \texttt{\string\fermata} 正确，\texttt{\string\fermate} 错误。
\item \texttt{\string\relative} 模式中，音符八度根据前一个音符自动推算。
\end{enumerate}
